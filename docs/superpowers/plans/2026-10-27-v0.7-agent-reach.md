# v0.7 agent reach: implementation plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make the agent in the LoadingAI pane see the model (view snapshots), name things right and reuse element lists (project index, `resolve_names`, `@rN` result sets), and remember and repeat work (approved project notes, playbooks).

**Architecture:** Pure logic goes in `LoadingAI.Core` (netstandard2.0), unit-tested with xUnit on Linux. Revit calls go through new `IRevitBridge` members implemented in `src/LoadingAI/Revit/AdapterRevitBridge*.cs` and faked in `src/LoadingAI.Tests/Fakes/FakeRevitBridge.cs`. Tools that only need the bridge live in `Plugins/core-tools`. Tools that need host state (notes, playbooks) are host tools registered by `ChatSession.RebuildRegistry`. The pane gets new protocol messages, and each one is added in three places: `HostProtocol.cs`, `web/src/bridge.ts` and the contract test. Every write still goes through `GatePolicy.Decide` and the one review card (ADR-0017).

**Tech stack:** C# 12 (net48 and net8 via Revit 2024–2027 targets), Revit API, xUnit, React 18 + TypeScript + vitest, Vite.

**Spec:** [docs/superpowers/specs/2026-10-27-v0.7-agent-reach-design.md](../specs/2026-10-27-v0.7-agent-reach-design.md) (revision 2).

```csharp
namespace LoadingAI.Llm;

/// <summary>Whether a provider's model can read images (P1). The user's switch wins; otherwise a name guess. Pure.</summary>
public static class VisionSupport
{
    private static readonly string[] Hints =
    {
        "gpt-4o", "gpt-4.1", "gpt-5", "o3", "o4", "claude", "gemini", "-vl", "vl-", "vision", "llava", "pixtral", "grok-4"
    };

    public static bool Supports(ProviderConfig? p)
    {
        if (p == null) return false;
        if (p.SupportsImages.HasValue) return p.SupportsImages.Value;
        var m = (p.Model ?? "").ToLowerInvariant();
        if (m.Length == 0) return false;
        foreach (var h in Hints)
            if (m.Contains(h)) return true;
        return false;
    }
}
```

```csharp
namespace LoadingAI.Revit;

/// <summary>Where view snapshots go and how many are kept (P1). Pure apart from file I/O.</summary>
public static class SnapshotFiles
{
    public const int Keep = 50;

    public static string DefaultDir => Path.Combine(
        Environment.GetFolderPath(Environment.SpecialFolder.ApplicationData), "LoadingAI", "snapshots");

    public static string NewPath(string dir, DateTime utcNow)
        => Path.Combine(dir, $"{utcNow:yyyyMMdd-HHmmss}-{Guid.NewGuid().ToString("N").Substring(0, 6)}.png");

    public static void Prune(string dir, int keep = Keep)
    {
        try
        {
            var files = new DirectoryInfo(dir).GetFiles("*.png").OrderByDescending(f => f.LastWriteTimeUtc).ToList();
            foreach (var f in files.Skip(keep))
                try { f.Delete(); } catch { }
        }
        catch { }
    }
}
```

```csharp
                        tx.Commit();

                    {
                        ExportRange = ExportRange.SetOfViews,
                        FilePath = Path.Combine(Path.GetDirectoryName(path)!, Path.GetFileNameWithoutExtension(path)),

                        ShadowViewsFileType = ImageFileType.PNG,

                        ZoomType = ZoomType.FitToPage,
                        PixelSize = size,
                        FitDirection = FitDirectionType.Horizontal

                    opts.SetViewsAndSheets(new List<ElementId> { target.Id });
                    doc.ExportImage(opts);
                }
                finally
                {
                    if (group != null) { try { group.RollBack(); } catch { } group.Dispose(); }
                }

            var (w, h) = PngSize(written.FullName);
            return new SnapshotDto(written.FullName, w, h, target.Name);

    private static (int, int) PngSize(string path)
    {
        try
        {
            using var fs = File.OpenRead(path);
            var b = new byte[24];
            if (fs.Read(b, 0, 24) < 24) return (0, 0);
            int W(int o) => (b[o] << 24) | (b[o + 1] << 16) | (b[o + 2] << 8) | b[o + 3];
            return (W(16), W(20));
        }
        catch { return (0, 0); }
    }
```

```csharp
public enum NameKind { Category, Level, Family, Type, Parameter, View, Sheet }

public sealed record NameCandidate(string Name, NameKind Kind, double Score, bool Exact);

/// <summary>Maps a user's word to the exact Revit names it most likely means (P2). Pure.</summary>
public static class NameResolver
{
    public const int MaxCandidates = 5;
    private const double MinScore = 0.45;

    private static readonly Dictionary<string, string[]> CategorySynonyms = new(StringComparer.OrdinalIgnoreCase)
    {
        ["beam"] = new[] { "Structural Framing" },
        ["framing"] = new[] { "Structural Framing" },
        ["column"] = new[] { "Structural Columns", "Columns" },
        ["slab"] = new[] { "Floors" },
        ["floor"] = new[] { "Floors" },
        ["wall"] = new[] { "Walls" },
        ["door"] = new[] { "Doors" },
        ["window"] = new[] { "Windows" },
        ["duct"] = new[] { "Ducts" },
        ["pipe"] = new[] { "Pipes" },
        ["room"] = new[] { "Rooms" },
        ["sheet"] = new[] { "Sheets" },
        ["view"] = new[] { "Views" },
        ["roof"] = new[] { "Roofs" },
        ["stair"] = new[] { "Stairs" },
        ["foundation"] = new[] { "Structural Foundations" },
    };

    public static IReadOnlyList<NameCandidate> Match(string term, IEnumerable<string> names, NameKind kind)
    {
        var t = Normalize(term);
        if (t.Length == 0) return Array.Empty<NameCandidate>();
        var synonyms = kind == NameKind.Category ? SynonymsFor(t) : Array.Empty<string>();
        var levelAlias = kind == NameKind.Level ? LevelAlias(t) : null;

        var scored = new List<NameCandidate>();
        foreach (var name in names.Distinct())
        {
            var n = Normalize(name);
            if (n.Length == 0) continue;
            double score;
            bool exact = false;
            if (string.Equals(term.Trim(), name, StringComparison.Ordinal)) { score = 1.0; exact = true; }
            else if (n == t) score = 0.98;
            else if (synonyms.Contains(name, StringComparer.OrdinalIgnoreCase)) score = 0.95;
            else if (levelAlias != null && n == levelAlias) score = 0.93;
            else if (n.StartsWith(t) || t.StartsWith(n)) score = 0.85 * Math.Min(t.Length, n.Length) / Math.Max(t.Length, n.Length) + 0.1;
            else
            {
                var overlap = TokenOverlap(t, n);
                var dist = Levenshtein(t, n);
                var edit = Math.Max(t.Length, n.Length) <= 12 && dist <= 2 ? 0.9 - dist * 0.1 : 0;
                score = Math.Max(overlap * 0.8, edit);
            }
            if (score >= MinScore) scored.Add(new NameCandidate(name, kind, Math.Round(score, 2), exact));
        }
        return scored.OrderByDescending(c => c.Score).ThenBy(c => c.Name.Length).Take(MaxCandidates).ToList();
    }

    public static string Normalize(string? s)
    {
        if (string.IsNullOrWhiteSpace(s)) return "";
        var sb = new StringBuilder();
        foreach (var raw in s!.Normalize(NormalizationForm.FormKC).ToLowerInvariant())
        {
            var ch = raw is '×' or '*' ? 'x' : raw;
            if (char.IsLetterOrDigit(ch)) sb.Append(ch);
            else if (sb.Length > 0 && sb[sb.Length - 1] != ' ') sb.Append(' ');
        }
        // "600 x 300mm" and "600x300" compare equal: drop spaces around x between digits, and a trailing unit.
        var text = System.Text.RegularExpressions.Regex.Replace(sb.ToString().Trim(), @"(\d)\s*x\s*(\d)", "$1x$2");
        text = System.Text.RegularExpressions.Regex.Replace(text, @"(\d)\s*mm\b", "$1");
        return text;
    }

    private static string[] SynonymsFor(string t)
    {
        var single = t.EndsWith("s") ? t.Substring(0, t.Length - 1) : t;
        return CategorySynonyms.TryGetValue(single, out var s) ? s
             : CategorySynonyms.TryGetValue(t, out s) ? s
             : Array.Empty<string>();
    }

    private static string? LevelAlias(string t)
    {
        var m = System.Text.RegularExpressions.Regex.Match(t, @"^(?:l|lvl|level)\s*(\d+)$");
        return m.Success ? "level " + m.Groups[1].Value : null;
    }

    private static double TokenOverlap(string a, string b)
    {
        var ta = a.Split(' ');
        var tb = b.Split(' ');
        int hits = ta.Count(x => tb.Any(y => y.StartsWith(x) || x.StartsWith(y)));
        return (double)hits / Math.Max(ta.Length, tb.Length);
    }

    private static int Levenshtein(string a, string b)
    {
        var d = new int[a.Length + 1, b.Length + 1];
        for (int i = 0; i <= a.Length; i++) d[i, 0] = i;
        for (int j = 0; j <= b.Length; j++) d[0, j] = j;
        for (int i = 1; i <= a.Length; i++)
            for (int j = 1; j <= b.Length; j++)
                d[i, j] = Math.Min(Math.Min(d[i - 1, j] + 1, d[i, j - 1] + 1), d[i - 1, j - 1] + (a[i - 1] == b[j - 1] ? 0 : 1));
        return d[a.Length, b.Length];
    }
}
```

```csharp
namespace LoadingAI.Tools;

/// <summary>Element-id lists a chat saved as @r1, @r2… (P2). Per chat session; Clear session empties it.</summary>
public sealed class ResultSetStore
{
    public const int MaxSets = 20;
    public const int MaxTotalIds = 50_000;

    private readonly List<(int N, IReadOnlyList<string> Ids)> _sets = new();
    private int _next = 1;
    private readonly object _lock = new();

    /// <summary>Saves the ids (deduplicated, order kept) and returns "@rN", or null when there are none.</summary>
    public string? Save(IReadOnlyList<string> ids)
    {
        var clean = ids.Where(i => !string.IsNullOrWhiteSpace(i)).Distinct().Take(MaxTotalIds).ToList();
        if (clean.Count == 0) return null;
        lock (_lock)
        {
            var n = _next++;
            _sets.Add((n, clean));
            while (_sets.Count > MaxSets || (_sets.Count > 1 && _sets.Sum(s => s.Ids.Count) > MaxTotalIds))
                _sets.RemoveAt(0);
            return "@r" + n;
        }
    }

    public bool TryGet(string handle, out IReadOnlyList<string> ids)
    {
        ids = Array.Empty<string>();
        if (handle == null || !handle.StartsWith("@r") || !int.TryParse(handle.Substring(2), out var n)) return false;
        lock (_lock)
        {
            foreach (var s in _sets)
                if (s.N == n) { ids = s.Ids; return true; }
        }
        return false;
    }

    public IReadOnlyList<string> Handles
    {
        get { lock (_lock) return _sets.Select(s => $"@r{s.N} ({s.Ids.Count})").ToList(); }
    }

    public void Clear() { lock (_lock) { _sets.Clear(); _next = 1; } }
}
```

```csharp
using System.Text;
using System.Text.Json;

namespace LoadingAI.Tools;

/// <summary>
/// Replaces "@rN" in elementIds / elementId with the saved ids before arguments are coerced and checked (P2),
/// so the review card and the tool see real ids. Only those two keys at the top level are touched.
/// </summary>
public static class ResultSetExpander
{
    public static bool TryExpand(string argsJson, ResultSetStore store, out string expanded, out string error)
    {
        expanded = argsJson;
        error = "";
        if (string.IsNullOrWhiteSpace(argsJson) || !argsJson.Contains("@r")) return true;
        JsonDocument doc;
        try { doc = JsonDocument.Parse(argsJson); } catch { return true; }
        using (doc)
        {
            if (doc.RootElement.ValueKind != JsonValueKind.Object) return true;
            var changed = false;
            using var ms = new MemoryStream();
            using (var w = new Utf8JsonWriter(ms))
            {
                w.WriteStartObject();
                foreach (var p in doc.RootElement.EnumerateObject())
                {
                    if (p.NameEquals("elementIds") && (p.Value.ValueKind == JsonValueKind.Array || IsHandle(p.Value)))
                    {
                        var items = p.Value.ValueKind == JsonValueKind.Array ? p.Value.EnumerateArray().ToList() : new List<JsonElement> { p.Value };
                        var outIds = new List<string>();
                        foreach (var it in items)
                        {
                            var s = it.ValueKind == JsonValueKind.String ? it.GetString() ?? "" : it.GetRawText();
                            if (IsHandle(s))
                            {
                                if (!store.TryGet(s, out var ids)) { error = Unknown(s, store); return false; }
                                outIds.AddRange(ids);
                                changed = true;
                            }
                            else outIds.Add(s);
                        }
                        w.WritePropertyName("elementIds");
                        w.WriteStartArray();
                        foreach (var id in outIds.Distinct()) w.WriteStringValue(id);
                        w.WriteEndArray();
                    }
                    else if (p.NameEquals("elementId") && IsHandle(p.Value))
                    {
                        var h = p.Value.GetString()!;
                        if (!store.TryGet(h, out var ids)) { error = Unknown(h, store); return false; }
                        if (ids.Count != 1) { error = $"{h} holds {ids.Count} elements; elementId takes one. Pass it as elementIds or pick one id."; return false; }
                        w.WriteString("elementId", ids[0]);
                        changed = true;
                    }
                    else p.WriteTo(w);
                }
                w.WriteEndObject();
            }
            if (changed) expanded = Encoding.UTF8.GetString(ms.ToArray());
            return true;
        }
    }

    private static bool IsHandle(JsonElement e) => e.ValueKind == JsonValueKind.String && IsHandle(e.GetString());
    private static bool IsHandle(string? s) => s != null && s.Length > 2 && s.StartsWith("@r") && s.Substring(2).All(char.IsDigit);

    private static string Unknown(string handle, ResultSetStore store)
    {
        var saved = store.Handles;
        return saved.Count == 0
            ? $"{handle} is not a saved result set. No sets are saved in this chat."
            : $"{handle} is not a saved result set. Saved: {string.Join(", ", saved)}.";
    }
}
```

```csharp
/// <summary>Project and global notes the agent reads at the start of each chat (P3, ADR-0018). One JSON file per key.</summary>
public sealed class NoteStore
{
    public const string GlobalKey = "global";
    public const int MaxNoteChars = 300;

    private readonly string _dir;
    private readonly object _lock = new();
    private readonly List<string> _issues = new();

    public NoteStore(string? dir = null)
        => _dir = dir ?? Path.Combine(Environment.GetFolderPath(Environment.SpecialFolder.ApplicationData), "LoadingAI", "memory");

    public IReadOnlyList<string> Issues { get { lock (_lock) return _issues.ToList(); } }

    public static string KeyFor(ProjectIdentityDto id)
    {
        var basis = !string.IsNullOrWhiteSpace(id.CloudModelGuid) ? "cloud:" + id.CloudModelGuid!.ToLowerInvariant()
            : !string.IsNullOrWhiteSpace(id.CentralPath) ? "central:" + id.CentralPath!.ToLowerInvariant()
            : "file:" + (id.PathName ?? "").ToLowerInvariant() + "|" + id.ProjectInfoUniqueId;
        using var sha = SHA256.Create();
        var hash = sha.ComputeHash(Encoding.UTF8.GetBytes(basis));
        return string.Concat(hash.Take(8).Select(b => b.ToString("x2")));
    }

    public IReadOnlyList<Note> List(string key) { lock (_lock) return Load(key); }

    public Note Add(string key, string text, string source)
    {
        var clean = UserInstructions.Sanitize(text)?.Replace('\n', ' ');
        if (string.IsNullOrWhiteSpace(clean)) throw new ArgumentException("A note needs text.");
        if (clean!.Length > MaxNoteChars) clean = clean.Substring(0, MaxNoteChars);
        var note = new Note(Guid.NewGuid().ToString("N").Substring(0, 8), clean, DateTime.UtcNow, source == "agent" ? "agent" : "user");
        lock (_lock) { var all = Load(key).ToList(); all.Add(note); Save(key, all); }
        return note;
    }

    public bool Update(string key, string id, string text)
    {
        var clean = UserInstructions.Sanitize(text)?.Replace('\n', ' ');
        if (string.IsNullOrWhiteSpace(clean)) return false;
        lock (_lock)
        {
            var all = Load(key).ToList();
            var i = all.FindIndex(n => n.Id == id);
            if (i < 0) return false;
            all[i] = all[i] with { Text = clean!.Length > MaxNoteChars ? clean.Substring(0, MaxNoteChars) : clean, Source = "user" };
            Save(key, all);
            return true;
        }
    }

    public bool Delete(string key, string id)
    {
        lock (_lock)
        {
            var all = Load(key).ToList();
            var removed = all.RemoveAll(n => n.Id == id) > 0;
            if (removed) Save(key, all);
            return removed;
        }
    }

    public void Clear(string key) { lock (_lock) Save(key, new List<Note>()); }

    private string PathFor(string key) => Path.Combine(_dir, key + ".json");

    private IReadOnlyList<Note> Load(string key)
    {
        var path = PathFor(key);
        if (!File.Exists(path)) return Array.Empty<Note>();
        try { return JsonSerializer.Deserialize<List<Note>>(File.ReadAllText(path)) ?? new List<Note>(); }
        catch
        {
            try { File.Copy(path, path + ".bad", true); File.Delete(path); } catch { }
            _issues.Add($"Notes file {key}.json could not be read and was kept as {key}.json.bad.");
            return Array.Empty<Note>();
        }
    }

    private void Save(string key, List<Note> notes)
    {
        Directory.CreateDirectory(_dir);
        var tmp = PathFor(key) + ".tmp";
        File.WriteAllText(tmp, JsonSerializer.Serialize(notes));
        if (File.Exists(PathFor(key))) File.Delete(PathFor(key));
        File.Move(tmp, PathFor(key));
    }
```

```csharp
namespace LoadingAI.Llm;

public sealed record Playbook(string Name, string Description, IReadOnlyList<string> Tools, string? Mode, string Body, string Source, string File);

/// <summary>Parses one playbook file: a "---" header (name, description, tools, mode) and a body (P3, ADR-0018). Pure.</summary>
public static class PlaybookParser
{
    public const int MaxBodyChars = 8000;
    private static readonly Regex NameRule = new("^[a-z0-9][a-z0-9-]{0,39}$");

    public static Playbook? Parse(string text, string file, string source, out string? error)
    {
        error = null;
        var t = (text ?? "").Replace("\r\n", "\n");
        if (!t.StartsWith("---\n")) { error = $"{System.IO.Path.GetFileName(file)}: no '---' header."; return null; }
        var end = t.IndexOf("\n---", 4, StringComparison.Ordinal);
        if (end < 0) { error = $"{System.IO.Path.GetFileName(file)}: header not closed with '---'."; return null; }
        var header = t.Substring(4, end - 4);
        var body = t.Substring(Math.Min(t.Length, end + 4)).Trim();
        var map = new Dictionary<string, string>(StringComparer.OrdinalIgnoreCase);
        foreach (var line in header.Split('\n'))
        {
            var i = line.IndexOf(':');
            if (i > 0) map[line.Substring(0, i).Trim()] = line.Substring(i + 1).Trim();
        }
        var name = map.TryGetValue("name", out var n) ? n : "";
        var fileName = System.IO.Path.GetFileName(file);
        if (!NameRule.IsMatch(name)) { error = $"{fileName}: name must be lower-case letters, digits and '-'."; return null; }
        if (!map.TryGetValue("description", out var desc) || desc.Length == 0) { error = $"{fileName}: description is missing."; return null; }
        string? mode = map.TryGetValue("mode", out var m) && m.Length > 0 ? m.ToLowerInvariant() : null;
        if (mode != null && mode != "plan" && mode != "build" && mode != "ask") { error = $"{fileName}: mode must be ask, plan or build."; return null; }
        if (body.Length == 0) { error = $"{fileName}: the playbook has no steps."; return null; }
        if (body.Length > MaxBodyChars) body = body.Substring(0, MaxBodyChars);
        var tools = map.TryGetValue("tools", out var tl)
            ? tl.Trim('[', ']').Split(',').Select(x => x.Trim().Trim('"', '\'')).Where(x => x.Length > 0).ToList()
            : new List<string>();
        return new Playbook(name, desc, tools, mode, body, source, file);
    }

    /// <summary>The user message a /name command sends.</summary>
    public static string Compose(Playbook pb, string? extra)
    {
        var sb = new StringBuilder($"Follow the playbook \"{pb.Name}\":\n{pb.Body}");
        if (!string.IsNullOrWhiteSpace(extra)) sb.Append("\n\nUser's note for this run: ").Append(extra!.Trim());
        return sb.ToString();
    }
}

/// <summary>Starters (read-only, shipped) then the user's folder; a user file with the same name wins.</summary>
public sealed class PlaybookCatalog
{
    public const int MaxListed = 30;

    public IReadOnlyList<Playbook> Items { get; }
    public IReadOnlyList<string> Issues { get; }

    private PlaybookCatalog(IReadOnlyList<Playbook> items, IReadOnlyList<string> issues) { Items = items; Issues = issues; }

    public static string DefaultUserDir => System.IO.Path.Combine(
        Environment.GetFolderPath(Environment.SpecialFolder.ApplicationData), "LoadingAI", "Playbooks");

    public static PlaybookCatalog Load(string? starterDir, string? userDir)
    {
        var byName = new Dictionary<string, Playbook>(StringComparer.OrdinalIgnoreCase);
        var issues = new List<string>();
        void Scan(string? dir, string source)
        {
            if (string.IsNullOrEmpty(dir) || !System.IO.Directory.Exists(dir)) return;
            foreach (var f in System.IO.Directory.GetFiles(dir, "*.md").OrderBy(x => x, StringComparer.OrdinalIgnoreCase))
            {
                string text;
                try { text = System.IO.File.ReadAllText(f); } catch (Exception ex) { issues.Add($"{System.IO.Path.GetFileName(f)}: {ex.Message}"); continue; }
                var pb = PlaybookParser.Parse(text, f, source, out var err);
                if (pb == null) { issues.Add(err!); continue; }
                byName[pb.Name] = pb;
            }
        }
        Scan(starterDir, "starter");
        Scan(userDir, "user");
        return new PlaybookCatalog(byName.Values.OrderBy(p => p.Name, StringComparer.Ordinal).ToList(), issues);
    }

    public Playbook? Get(string name) => Items.FirstOrDefault(p => string.Equals(p.Name, name, StringComparison.OrdinalIgnoreCase));

    /// <summary>The prompt list: names and descriptions only. The body is read with open_playbook.</summary>
    public string PromptList()
    {
        if (Items.Count == 0) return "";
        var sb = new StringBuilder("\n# Playbooks\nThe user's saved procedures. Read one with open_playbook(name) when the task matches.\n");
        foreach (var p in Items.Take(MaxListed)) sb.Append("- ").Append(p.Name).Append(": ").Append(p.Description).Append('\n');
        return sb.ToString();
    }
}
```

```ts
export type PlaybookInfo = { name: string; description: string; mode?: 'ask' | 'plan' | 'build' | null; source: string; tools: string[] };

export function filterCommandsWith(query: string, playbooks: PlaybookInfo[]): SlashCommand[] {
  const builtIn = filterCommands(query);
  const q = query.toLowerCase().trim().replace(/^\//, '');
  const taken = new Set(SLASH_COMMANDS.map(c => c.name));
  const pbs = playbooks
    .filter(p => !taken.has(p.name) && (!q || p.name.includes(q) || p.description.toLowerCase().includes(q)))
    .map(p => ({ name: p.name, label: `/${p.name}`, description: `Playbook · ${p.description}`, argsHint: '[note]', mode: p.mode ?? undefined }));
  return [...builtIn, ...pbs];
}

export function playbookCommand(input: string, playbooks: PlaybookInfo[]): { name: string; extra: string; mode: 'ask' | 'plan' | 'build' | null } | null {
  const m = input.trimStart().match(/^\/([a-z0-9][a-z0-9-]*)\s*([\s\S]*)$/);
  if (!m || SLASH_COMMANDS.some(c => c.name === m[1])) return null;
  const pb = playbooks.find(p => p.name === m[1]);
  return pb ? { name: pb.name, extra: m[2].trim(), mode: pb.mode ?? null } : null;
}
```
